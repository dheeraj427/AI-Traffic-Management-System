# ---------------- AUDIO FUNCTIONS (THREADED) ----------------

def speak_worker(text):
    try:
        filename = f"voice_{int(time.time() * 1000)}.mp3"
        tts = gTTS(text=text, lang='en')
        tts.save(filename)
        playsound(filename)
        if os.path.exists(filename):
            os.remove(filename)
    except Exception:
        pass

def speak(text):
    threading.Thread(target=speak_worker, args=(text,), daemon=True).start()

def play_beep_worker():
    try:
        print('\a')
    except Exception:
        pass

def play_beep():
    threading.Thread(target=play_beep_worker, daemon=True).start()

def announce_signal(lane, duration):
    friendly_name = LANE_NAMES.get(lane, lane)
    text = f"Signal update. {friendly_name} lane is now green for {duration} seconds."
    speak(text)

# ---------------- SIGNAL CALCULATION ----------------

def calculate_lane_times(lane_counts):
    base_cycle = BASE_GREEN * 4
    total_cars = sum(lane_counts.values())
    durations = {}

    if total_cars == 0:
        even = base_cycle // 4
        for lane in LANES:
            durations[lane] = min(even,MAX_GREEN)
        return durations

    weights = {lane:count/total_cars for lane,count in lane_counts.items()}

    for lane in LANES:
        t = int(base_cycle * weights[lane])
        t = max(MIN_GREEN, min(MAX_GREEN, t))
        durations[lane] = t

    return durations

# ---------------- BUILD SIGNAL SCHEDULE ----------------

def build_cycle_schedule(durations,start_time):
    schedule=[]
    current=start_time
    for lane in LANES:
        end=current+durations[lane]
        schedule.append({"lane":lane,"start":current,"end":end})
        current=end
    return schedule

            if current_time < item["start"]:
                lane_wait[lane]=int(item["start"]-current_time)

def lane_breakdown_ui(lane_counts, active_lane=None):

    col3, col4 = st.columns(2)

    def split(count):
        return {
            "car": int(count*0.4),
            "bike": int(count*0.3),
            "bus": int(count*0.2),
            "truck": int(count*0.1)
        }

    def render_card(title, count, is_active=False):
        data = split(count)
        status = "🟢 ACTIVE" if is_active else "🔴 STOPPED"
        wrapper_class = "pulse" if is_active else ""

        st.markdown(f"""
        <div class="glass-card glow-box fade-in {wrapper_class}" style="background:#0f172a; margin:10px; border-radius:15px; border:1px solid rgba(255,255,255,0.2);">
            <h4 style="color:white; margin-bottom: 10px;">{title} - <span style="font-size: 16px;">{status}</span></h4>
            <p style="color: #aaa; font-size: 18px;">🚗 {data['car']} | 🏍️ {data['bike']} | 🚌 {data['bus']} | 🚚 {data['truck']}</p>
        </div>
        """,unsafe_allow_html=True)

    with col1:
        render_card("NORTH", lane_counts[lanes[0]], is_active=(lanes[0] == active_lane))
    with col2:
        render_card("SOUTH", lane_counts[lanes[1]], is_active=(lanes[1] == active_lane))
    with col3:
        render_card("EAST", lane_counts[lanes[2]], is_active=(lanes[2] == active_lane))

# ---------------- TRAFFIC PREDICTION ----------------

def predict_traffic(history):
    if len(history)<5:
        return history[-1]
    X=np.arange(len(history)).reshape(-1,1)
    y=np.array(history)
    model_lr=LinearRegression()
    model_lr.fit(X,y)
    future=np.array([[len(history)+1]])
    return int(model_lr.predict(future)[0])
